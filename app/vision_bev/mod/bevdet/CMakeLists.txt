cmake_minimum_required(VERSION 3.5.0)
if(NOT "${SAMPLE_BUILD_TARGET}" STREQUAL "sdk")
project(sample_bevdet)
endif()

set(NPU_SAMPLE_ROOT_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../../../../)
include(${NPU_SAMPLE_ROOT_DIR}/cmake/npu_sample_build.cmake)

file(GLOB NPU_SAMPLE_FASTBEV_SRCS   ${CMAKE_CURRENT_SOURCE_DIR}/*.c*)
file(GLOB NPU_SAMPLE_BEV_INIT_SRCS  ${CMAKE_CURRENT_SOURCE_DIR}/../common/data_init/*.c*)
file(GLOB NPU_SAMPLE_BEV_HEAD_SRCS  ${CMAKE_CURRENT_SOURCE_DIR}/../common/center_head/*.c*)
file(GLOB NPU_SAMPLE_BEV_VIEW_SRCS  ${CMAKE_CURRENT_SOURCE_DIR}/../common/view/*.c*)
set(NPU_SAMPLE_FASTBEV_INC_PATH     ${NPU_SAMPLE_ROOT_DIR}/third_party/json/include)
set(NPU_SAMPLES_BEV_COMMON_DIR      ${CMAKE_CURRENT_SOURCE_DIR}/../common)

# sample_bevdet
build_binary(
    TARGET   linux_npu_sample_bevdet
    SRCS     ${NPU_SAMPLE_COMMOM_SRCS} ${NPU_SAMPLE_FASTBEV_SRCS} ${NPU_SAMPLE_BEV_INIT_SRCS} ${NPU_SAMPLE_BEV_HEAD_SRCS} ${NPU_SAMPLE_BEV_VIEW_SRCS}
    INC_DIRS ${NPU_SAMPLE_INC_PATH} ${OPENCV_INCLUDE_DIR} ${NPU_SAMPLE_FASTBEV_INC_PATH} ${NPU_SAMPLES_BEV_COMMON_DIR}
    LIB_DIRS ${NPU_SAMPLE_LIB_DIRS} ${OPENCV_OUT_DIR}
    LIBS     ${NPU_SAMPLE_LIB_NAME} ${OPENCV_LIBS_NAME}
    OUT_NAME ${NPU_SAMPLE_BEVDEV_NAME}
    OUT_DIR  ${NPU_SAMPLE_OUT_DIR}
    DEFS LINUX
)
