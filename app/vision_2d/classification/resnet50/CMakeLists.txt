cmake_minimum_required(VERSION 3.5.0)
if(NOT "${SAMPLE_BUILD_TARGET}" STREQUAL "sdk")
project(sample_resnet50)
endif()

set(NPU_SAMPLE_ROOT_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../../../../)
include(${NPU_SAMPLE_ROOT_DIR}/cmake/npu_sample_build.cmake)

file(GLOB NPU_SAMPLE_RESNET50_SRCS  ${CMAKE_CURRENT_SOURCE_DIR}/*.c*)
file(GLOB NPU_SAMPLE_COMMON_INIT_SRCS  ${CMAKE_CURRENT_SOURCE_DIR}/../common/*.c*)
set(NPU_SAMPLE_RESNET50_INC_PATH ${CMAKE_CURRENT_SOURCE_DIR}/../common)

# sample_resnet50
build_binary(
    TARGET   linux_npu_sample_resnet50
    SRCS     ${NPU_SAMPLE_COMMOM_SRCS} ${NPU_SAMPLE_RESNET50_SRCS} ${NPU_SAMPLE_COMMON_INIT_SRCS}
    INC_DIRS ${NPU_SAMPLE_INC_PATH} ${NPU_SAMPLE_RESNET50_INC_PATH}
    LIB_DIRS ${NPU_SAMPLE_LIB_DIRS}
    LIBS     ${NPU_SAMPLE_LIB_NAME}
    OUT_NAME ${NPU_SAMPLE_RESNET50_NAME}
    OUT_DIR  ${NPU_SAMPLE_OUT_DIR}
    DEFS LINUX
)
