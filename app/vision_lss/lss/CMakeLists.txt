cmake_minimum_required(VERSION 3.5)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(NOT "${SAMPLE_BUILD_TARGET}" STREQUAL "sdk")
project(LSS)
endif()

set(NPU_SAMPLE_ROOT_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../../..)
include(${NPU_SAMPLE_ROOT_DIR}/cmake/npu_sample_build.cmake)

file(GLOB NPU_SAMPLE_LSS_SRCS  ${CMAKE_CURRENT_SOURCE_DIR}/*.c*)
file(GLOB NPU_SAMPLE_LSS_EVALUATE_SRCS  ${CMAKE_CURRENT_SOURCE_DIR}/../common/evaluate/*.c*)

set(NPU_SAMPLE_LSS_INC_PATH    ${NPU_SAMPLE_ROOT_DIR}/third_party/json/include)
set(NPU_SAMPLES_LSS_COMMON_DIR         ${CMAKE_CURRENT_SOURCE_DIR}/../common)

build_binary(
    TARGET   linux_npu_sample_LSS
    SRCS     ${NPU_SAMPLE_COMMOM_SRCS} ${NPU_SAMPLE_LSS_SRCS} ${NPU_SAMPLE_LSS_EVALUATE_SRCS} ${NPU_SAMPLE_LSS_INC_PATH}
    INC_DIRS ${NPU_SAMPLE_INC_PATH}
    LIB_DIRS ${NPU_SAMPLE_LIB_DIRS}
    LIBS     ${NPU_SAMPLE_LIB_NAME}
    OUT_NAME ${NPU_SAMPLE_LSS_NAME}
    OUT_DIR  ${NPU_SAMPLE_OUT_DIR}
    DEFS LINUX
)
